cmake_minimum_required(VERSION 3.20)
project(tiermem LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()

option(TIERMEM_SANITIZE "Compilar con AddressSanitizer + UBSan" OFF)
option(TIERMEM_TSAN "Compilar con ThreadSanitizer" OFF)
option(TIERMEM_USE_LIBURING "Usar liburing si está disponible" ON)
option(TIERMEM_WITH_CUFILE "Backend cuFile/GDS (NO VERIFICADO, requiere CUDA)" OFF)

add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(TIERMEM_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=all)
  add_link_options(-fsanitize=address,undefined)
endif()

if(TIERMEM_TSAN)
  add_compile_options(-fsanitize=thread)
  add_link_options(-fsanitize=thread)
endif()

find_package(Threads REQUIRED)
add_library(tiermem src/tiermem.cpp src/backend.cpp)
target_include_directories(tiermem PUBLIC include)
target_link_libraries(tiermem PUBLIC Threads::Threads)

if(TIERMEM_USE_LIBURING)
  find_path(LIBURING_INCLUDE liburing.h)
  find_library(LIBURING_LIB uring)
  if(LIBURING_INCLUDE AND LIBURING_LIB)
    target_compile_definitions(tiermem PRIVATE TIERMEM_HAVE_LIBURING=1)
    target_include_directories(tiermem PRIVATE ${LIBURING_INCLUDE})
    target_link_libraries(tiermem PRIVATE ${LIBURING_LIB})
    message(STATUS "tiermem: liburing encontrado -> backend io_uring habilitado")
  else()
    message(STATUS "tiermem: liburing NO encontrado -> sólo pread+O_DIRECT")
  endif()
endif()

if(TIERMEM_WITH_CUFILE)
  find_path(CUFILE_INCLUDE cufile.h)
  if(NOT CUFILE_INCLUDE)
    message(FATAL_ERROR "TIERMEM_WITH_CUFILE=ON pero no se encontró cufile.h (backend NO VERIFICADO)")
  endif()
  target_sources(tiermem PRIVATE src/backend_cufile.cpp)
  target_compile_definitions(tiermem PRIVATE TIERMEM_WITH_CUFILE=1)
endif()

enable_testing()
add_executable(tiermem_tests tests/tests.cpp)
target_link_libraries(tiermem_tests PRIVATE tiermem)
foreach(t backend_pread backend_iouring fault_secuencial fault_aleatorio cola_no_multiplo
          expulsion_bajo_presion nivel_host prefetch_secuencial prefetch_explicito multihilo
          multihilo_presion varios_tensores solo_lectura_sigsegv syscall_sobre_rango sin_nivel_host)
  add_test(NAME ${t} COMMAND tiermem_tests ${t})
  set_tests_properties(${t} PROPERTIES TIMEOUT 120)
endforeach()

add_executable(tiermem_bench bench/bench.cpp)
target_link_libraries(tiermem_bench PRIVATE tiermem)
